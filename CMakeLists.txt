cmake_minimum_required(VERSION 3.16)
project(csvscope VERSION 0.1.0 LANGUAGES CXX)
add_executable(csvscope src/main.cpp)
target_compile_features(csvscope PRIVATE cxx_std_17)
if(MSVC)
  target_compile_options(csvscope PRIVATE /W4)
else()
  target_compile_options(csvscope PRIVATE -Wall -Wextra -Wpedantic)
endif()
include(CTest)
if(BUILD_TESTING)
  find_package(Python3 COMPONENTS Interpreter)
  if(Python3_FOUND)
    add_test(NAME regression COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_cli.py $<TARGET_FILE:csvscope>)
  endif()
endif()
install(TARGETS csvscope RUNTIME DESTINATION bin)
